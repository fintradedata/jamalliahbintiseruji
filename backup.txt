    const password = document.getElementById("password");

<script>

let actionType = "";

function openModal(type) {

    actionType = type;

    const modal = document.getElementById("modal");
    const title = document.getElementById("modalTitle");
    const description = document.getElementById("modalDescription");

    if(type === "deposit") {

        title.textContent = "Penyimpanan Dana";
        description.textContent =
            "Masukkan jumlah dana yang ingin disimpan.";

    } else {

        title.textContent = "Penarikan Dana";
        description.textContent =
            "Masukkan jumlah dana yang ingin ditarik.";

    }

    modal.style.display = "flex";
}

function closeModal() {

    document.getElementById("modal").style.display = "none";

    document.getElementById("amount").value = "";
}

function processAction() {

    const amount =
        document.getElementById("amount").value;

    if(!amount || Number(amount) <= 0) {

        alert("Masukkan jumlah yang valid.");
        return;

    }

    if(actionType === "deposit") {

        alert(
            "Demo: permintaan penyimpanan sebesar $" +
            Number(amount).toLocaleString() +
            " dibuat."
        );

    } else {

        alert(
            "Demo: permintaan penarikan sebesar $" +
            Number(amount).toLocaleString() +
            " dibuat."
        );

    }

    closeModal();
}

</script>

/*

function generateTransactionId(){

    const now = new Date();

    const date =
        now.getFullYear() +
        String(now.getMonth() + 1).padStart(2,"0") +
        String(now.getDate()).padStart(2,"0");

    const random =
        Math.floor(100000 + Math.random() * 900000);

    return `WD-${date}-${random}`;
}


/*
=========================================
AMBIL / BUAT ID TRANSAKSI
=========================================
*/

function getTransactionId(){

    let transactionId =
        localStorage.getItem("withdrawalTransactionId");

    if(!transactionId){

        transactionId =
            generateTransactionId();

        localStorage.setItem(
            "withdrawalTransactionId",
            transactionId
        );
    }

    return transactionId;
}


/*
=========================================
UPDATE WAKTU REALTIME
=========================================
*/

function updateRealtime(){

    const now = new Date();

    const formatted =
        new Intl.DateTimeFormat("id-ID",{

            day:"2-digit",
            month:"long",
            year:"numeric",

            hour:"2-digit",
            minute:"2-digit",
            second:"2-digit",

            hour12:false

        }).format(now);

    document.getElementById(
        "transactionDate"
    ).textContent = formatted;
}


/*
=========================================
INITIALIZE
=========================================
*/

document.getElementById(
    "transactionId"
).textContent = getTransactionId();

updateRealtime();


/*
=========================================
UPDATE SETIAP 1 DETIK
=========================================
*/

setInterval(updateRealtime,1000);

        history.back();

    }else{

}
